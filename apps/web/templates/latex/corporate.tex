%------------------------------------------------------------------
% Rules and section headings
% The heading text and its rule are one unbreakable box, so a page break can
% never strand a heading (or its rule) away from the content below it. The
% first heading has no rule of its own: the header rule sits directly above it.
%------------------------------------------------------------------
\newcommand{\rRuleBox}{\hbox to\linewidth{\hss\vrule width \dimexpr\linewidth+3pt\relax height 0.5pt\hss}}
\newif\ifrFirstSection \rFirstSectiontrue
\newif\ifrAfterHeading \rAfterHeadingfalse
\newcommand{\rSectionTitle}[1]{%
  \global\rAfterHeadingtrue
  \ifrFirstSection
    \global\rFirstSectionfalse
    \vbox{\hbox to\linewidth{\smash{#1:}\vphantom{A}\hfil}\vskip 3.16pt}%
  \else
    \vbox{\hbox to\linewidth{\smash{#1:}\vphantom{A}\hfil}%
          \nointerlineskip\vskip 3.9pt\rRuleBox}%
  \fi}
\titleformat{\section}
  {\normalsize\bfseries\color{accent}}{}{0em}{\rSectionTitle}
